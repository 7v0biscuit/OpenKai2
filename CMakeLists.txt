get_directory_property(probe_targets BUILDSYSTEM_TARGETS)
foreach(target IN LISTS probe_targets)
    target_compile_features(${target} PRIVATE cxx_std_17)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /EHsc)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()

add_library(openkai2_disc STATIC src/disc_image.cpp src/mods/afs.cpp src/mods/disc_overlay.cpp src/mods/mod_library.cpp src/mods/importer.cpp src/mods/roster.cpp src/mods/select_alternates.cpp src/mods/select_screen.cpp src/mods/disc_identity.cpp)
