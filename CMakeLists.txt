cmake_minimum_required(VERSION 3.16)
project(OpenKai2 LANGUAGES CXX)

# Standalone EE component probes. The game runtime is built by tools/native.
add_executable(openkai2_shared_entry src/probes/shared_entry.cpp)
add_executable(openkai2_tail_wrappers src/probes/tail_wrappers.cpp)
add_executable(openkai2_virtual_cp0 src/probes/virtual_cp0.cpp)
add_executable(openkai2_mmio_eret src/probes/mmio_eret.cpp)
add_executable(openkai2_wrapper_chain src/probes/wrapper_chain.cpp)
add_executable(openkai2_interrupt_transitions src/probes/interrupt_transitions.cpp)
add_executable(openkai2_cp0_observations src/probes/cp0_observations.cpp)
add_executable(openkai2_irq_handoff src/probes/irq_handoff.cpp)
add_executable(openkai2_handler_probe src/probes/handler_list_probe.cpp)
add_executable(openkai2_handler_bridge src/probes/handler_bridge.cpp)
add_executable(openkai2_helper_continuation src/probes/helper_continuation.cpp)
add_executable(openkai2_irq_handler src/probes/irq_handler.cpp)
add_executable(openkai2_list_walk src/probes/list_walk.cpp)
add_executable(openkai2_list_irq_chain src/probes/list_irq_chain.cpp)
add_executable(openkai2_live_guest_memory src/probes/live_guest_memory.cpp)
add_executable(openkai2_mapped_guest_memory src/probes/mapped_guest_memory.cpp)

get_directory_property(probe_targets BUILDSYSTEM_TARGETS)
foreach(target IN LISTS probe_targets)
    target_compile_features(${target} PRIVATE cxx_std_17)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /EHsc)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
